ADT stack i s elem + boolean + operations :
empty: −> stack
push : stack *elem −> stack
pop : stack −> stack
top : stack −> elem
isEmpty : stack −> booleanEquations :
isEmpty (empty) = true
isEmpty (push( s , e ) ) = false
pop(push( s , e ) ) = s
pop(empty) = error
top (push( s , e ) ) = e
top (empty) = error
